// Uso básico del módulo storage const productData = { name: "Manzanas Fuji", quantity: 5, expiryDate: "2024-12-31", category: "Frutas", notes: "Comprar en el mercado local" }; // Añadir producto const newProduct = window.productStorage.add(productData); // Obtener todos los productos const allProducts = window.productStorage.getAll(); // Buscar productos const searchResults = window.productStorage.search("manzana"); // Obtener estadísticas const stats = window.productStorage.getStatistics(); // Eliminar productos expirados const expiredProducts = window.productStorage.removeExpired();
